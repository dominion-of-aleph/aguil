  dotnet fsi --use:AvaloniaPlayground.Editor.Tests/Examples.fsx

  let res = AvaloniaPlayground.Editor.Styler.style parser grammar.Highlights "def foo(x) do x + 3 end ";;
  describe ("def foo(x) x + 3 end", res);;

  // Or equivalently
  let res = style "def foo(x) x + 3 end";;
  describe res;;
